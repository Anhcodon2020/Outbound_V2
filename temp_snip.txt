
        class Palletizer {
            constructor() {}
            
            // Hàm sinh mẫu lưới layer tối ưu (Chấp nhận Pallet Ratio +5% Overhang)
            calculateBestLayerPatterns(palletL, palletW, boxL, boxW) {
                let maxBoxes = 0;
                let bestLayouts = [];
                
                // +5% allowance for overhang
                const allowL = palletL * 1.05;
                const allowW = palletW * 1.05;

                const fillArea = (x0, y0, maxL, maxW, bl, bw, angle) => {
                    const boxes = [];
                    const nx = Math.floor(maxL / bl);
                    const ny = Math.floor(maxW / bw);
                    for (let i = 0; i < nx; i++) {
                        for (let j = 0; j < ny; j++) {
                            // Centered on its tile
                            boxes.push({ x: x0 + i * bl + bl / 2.0, y: y0 + j * bw + bw / 2.0, angle });
                        }
                    }
                    return boxes;
                };

                const step = Math.max(1, Math.floor(Math.min(boxL, boxW) / 2));
                const addLayout = (layout) => {
                    if (layout.length > maxBoxes) { maxBoxes = layout.length; bestLayouts = [layout]; } 
                    else if (layout.length === maxBoxes && maxBoxes > 0) { bestLayouts.push(layout); }
                };

                for (let xSplit = 0; xSplit <= allowL; xSplit += step) {
                    for (let leftRot of [false, true]) {
                        for (let rightRot of [false, true]) {
                            const leftBoxes = fillArea(0, 0, xSplit, allowW, leftRot?boxW:boxL, leftRot?boxL:boxW, leftRot ? 90 : 0);
                            const rightBoxes = fillArea(xSplit, 0, allowL - xSplit, allowW, rightRot?boxW:boxL, rightRot?boxL:boxW, rightRot ? 90 : 0);
                            addLayout([...leftBoxes, ...rightBoxes]);
                        }
                    }
                }
                for (let ySplit = 0; ySplit <= allowW; ySplit += step) {
                    for (let topRot of [false, true]) {
                        for (let btmRot of [false, true]) {
                            const btmBoxes = fillArea(0, 0, allowL, ySplit, btmRot?boxW:boxL, btmRot?boxL:boxW, btmRot ? 90 : 0);
                            const topBoxes = fillArea(0, ySplit, allowL, allowW - ySplit, topRot?boxW:boxL, topRot?boxL:boxW, topRot ? 90 : 0);
                            addLayout([...btmBoxes, ...topBoxes]);
                        }
                    }
                }
                return bestLayouts.length ? bestLayouts : [];
            }

            planMultipleOrders(orderList, allowMix, palletType = "1.2") {
                const palletLength = palletType === "1.9" ? 1900 : 1200;
                const palletWidth = 1000;
                const palletBaseHeight = 170;
                const maxHeightLimit = 2600;
                
                let pallets = [];
                let currentPallet = { boxes: [], zMax: palletBaseHeight, highestLayer: 0 };
                
                const pushPallet = () => {
                    if (currentPallet.boxes.length > 0) pallets.push(currentPallet);
                    currentPallet = { boxes: [], zMax: palletBaseHeight, highestLayer: 0 };
                };

                for (let order of orderList) {
                    if (!allowMix && currentPallet.boxes.length > 0) {
                        pushPallet();
                    }
                    
                    const patterns = this.calculateBestLayerPatterns(palletLength, palletWidth, order.l, order.w);
                    let remainQty = order.qty;
                    const allowL = palletLength * 1.05;
                    const allowW = palletWidth * 1.05;

                    while (remainQty > 0) {
                        let usableHeight = maxHeightLimit - currentPallet.zMax;
                        let generated = [];
                        
                        if (usableHeight >= order.h && patterns.length > 0) {
                            const maxByH = Math.floor(usableHeight / order.h);
                            const zStart = currentPallet.zMax;
                            let packed = 0;
                            
                            for (let layer = 0; layer < maxByH; layer++) {
                                if (packed >= remainQty) break;
                                const patIdx = layer % patterns.length;
                                const zCenter = zStart + (layer * order.h) + order.h / 2.0;
                                
                                for (let item of patterns[patIdx]) {
                                    if (packed >= remainQty) break;
                                    let cx = item.x, cy = item.y;
                                    if (patterns.length === 1 && layer % 2 === 1) { 
                                        cx = allowL - item.x; cy = allowW - item.y;
                                    }
                                    
                                    const cbm = (order.l * order.w * order.h) / 1000000000;
                                    
                                    generated.push({
                                        id: Date.now() + Math.random(),
                                        sku: order.sku, x: cx, y: cy, z: zCenter, angle: item.angle,
                                        layer: currentPallet.highestLayer + layer + 1,
                                        l: order.l, w: order.w, h: order.h, weight: order.weight, cbm: cbm
                                    });
                                    packed++;
                                }
                            }
                        }
                        
                        if (generated.length === 0) {
                            pushPallet();
                            if (currentPallet.boxes.length === 0 && maxHeightLimit - currentPallet.zMax < order.h) {
                                console.error("SKU is too tall to fit on a blank pallet");
                                break; 
                            }
                        } else {
                            currentPallet.boxes.push(...generated);
                            remainQty -= generated.length;
                            
                            for (let b of generated) {
                                let topEdge = b.z + (b.h / 2.0);
                                if (topEdge > currentPallet.zMax) currentPallet.zMax = topEdge;
                                if (b.layer > currentPallet.highestLayer) currentPallet.highestLayer = b.layer;
                            }
                        }
                    }
                }
                
                if (currentPallet.boxes.length > 0) pushPallet();
                return pallets;
            }
        }

                    this.inpH.value = (match.height * 10) || 200;

            findEmptySpotOnPallet(pIdx, boxL, boxW, boxH, pLen) {
                const step = 10;
                let bestZ = 999999;
                let bestSpot = { x: pLen/2, y: 500, z: 170 + boxH/2 };

                for(let cx = boxL/2; cx <= pLen - boxL/2 + 2; cx += step) {
                    for(let cy = boxW/2; cy <= 1000 - boxW/2 + 2; cy += step) {
                        
                        const tMinX = cx - boxL/2 + 1; const tMaxX = cx + boxL/2 - 1;
                        const tMinY = cy - boxW/2 + 1; const tMaxY = cy + boxW/2 - 1;
                        
                        let supportZ = 170; // ground floor
                        
                        for(let b of this.boxesData) {
                            if (b.pIdx !== pIdx || b.deleted) continue;
                            
                            const bw = b.angle === 0 ? b.w : b.l;
                            const bl = b.angle === 0 ? b.l : b.w;
                            const bMinX = b.x - bl/2; const bMaxX = b.x + bl/2;
                            const bMinY = b.y - bw/2; const bMaxY = b.y + bw/2;
                            
                            if (tMinX < bMaxX && tMaxX > bMinX && tMinY < bMaxY && tMaxY > bMinY) {
                                let bTop = b.z + b.h/2;
                                if (bTop > supportZ) supportZ = bTop;
                            }
                        }
                        
                        if (supportZ < bestZ) {
                            bestZ = supportZ;
                            bestSpot = { x: cx, y: cy, z: supportZ + boxH/2 };
                        }
                        
                        if (bestZ === 170) return bestSpot;
                    }
                }
                
                if (bestZ > 170) bestSpot.z += 2; // subtle float above boxes
                return bestSpot;
            },

            addManualBox() {
                let l = parseFloat(this.inpL.value) || 400; let w = parseFloat(this.inpW.value) || 300;
                let h = parseFloat(this.inpH.value) || 200; const wt = parseFloat(this.inpWeight.value) || 5;
                const sku = this.inpSku.value?.trim() || `MANUAL-${Math.floor(Math.random()*1000)}`;
                const cbm = (l * w * h) / 1000000000;
                const pLen = this.inpPallet.value === "1.9" ? 1900 : 1200;
                const orientEl = document.getElementById('inp-orient');
                const angle = Number.isFinite(this.dropPreviewAngle) ? this.dropPreviewAngle : (orientEl ? parseInt(orientEl.value) : 0);

                // Lật úp theo tùy chọn (flip chiều rộng hoặc chiều dài lên cao)
                if (this.dropPreviewFlip === 'flipW') [w, h] = [h, w];
                else if (this.dropPreviewFlip === 'flipL') [l, h] = [h, l];

                // Tính kích thước hiệu dụng tuỳ theo góc thả thủ công
                const quarterIdx = Math.abs(Math.round(angle / 90)) % 2;
                const effL = quarterIdx === 0 ? l : w;
                const effW = quarterIdx === 0 ? w : l;

                const spot = this.findEmptySpotOnPallet(0, effL, effW, h, pLen);

                const boxData = {
                    id: Date.now() + Math.random(),
                    sku: sku,
                    x: spot.x,
                    y: spot.y,     
                    z: spot.z, 
                    angle: angle,
                    layer: 0,
                    l: l, w: w, h: h, weight: wt, cbm: cbm,
                    pIdx: 0,     // assume pallet 0 for manual drops
                    offsetX: 0,
                    isManual: true
                };

                // Add to flattened array bypassing auto plan
                this.boxesData.push(boxData);
                this.drawNextCarton(true); // draw immediately
                this.activateCarton(boxData.id);
                this.lblStatus.innerText = 'Đã thả 1 kiện hàng thủ công. Kéo thả để di chuyển.';
            },

            addManualBoxAtPoint(worldPoint) {
                let l = parseFloat(this.inpL.value) || 400; let w = parseFloat(this.inpW.value) || 300;
                let h = parseFloat(this.inpH.value) || 200; const wt = parseFloat(this.inpWeight.value) || 5;
                const sku = this.inpSku.value?.trim() || `MANUAL-${Math.floor(Math.random()*1000)}`;
                const cbm = (l * w * h) / 1000000000;
                const pLen = this.inpPallet.value === "1.9" ? 1900 : 1200;
                const orientEl = document.getElementById('inp-orient');
                const angle = Number.isFinite(this.dropPreviewAngle) ? this.dropPreviewAngle : (orientEl ? parseInt(orientEl.value) : 0);
                const quarterIdx = Math.abs(Math.round(angle / 90)) % 2;

                if (this.dropPreviewFlip === 'flipW') [w, h] = [h, w];
                else if (this.dropPreviewFlip === 'flipL') [l, h] = [h, l];

                const boxData = {
                    id: Date.now() + Math.random(),
                    sku: sku,
                    x: worldPoint.x + (pLen/2),
                    y: worldPoint.z + 500,     
                    z: worldPoint.y + h/2, 
                    angle: angle,
                    layer: 0,
                    l: l, w: w, h: h, weight: wt, cbm: cbm,
                    pIdx: 0,
                    offsetX: 0,
                    isManual: true
                };

                // Add to flattened array bypassing auto plan
                this.boxesData.push(boxData);
                this.drawNextCarton(true); 
                
                // Snap using existing logic dynamically to prevent overlaps
                this.activeBoxId = boxData.id;
                this.snapActiveToPoint(worldPoint);
                this.deselectActive(); // Quickly deselect to allow more drops
            },

            activateCarton(boxId) {
                this.deselectActive();
                this.activeBoxId = boxId;
                
                const mesh = this.meshes.find(m => m.userData.id === boxId);
                const data = this.boxesData.find(b => b.id === boxId);
                if (!mesh || !data) return;

                document.getElementById('interactive-panel').style.display = 'block';
                document.getElementById('active-carton-title').innerText = `Đang chọn: ${data.sku}`;
                
                // Hỗ trợ hiển thị chọn đối với hình khối có nhiều nhãn dán (Array material)
                if (Array.isArray(mesh.material)) {
                    mesh.material.forEach(mat => mat.emissive.setHex(0x444444));
                } else {
                    mesh.material.emissive.setHex(0x444444);
                }

                // Highlight màu trên danh sách thanh công cụ
                const uiItem = document.getElementById(`ui-item-${boxId}`);
                if (uiItem) {
                    uiItem.classList.add('active');
                    uiItem.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
                }
                
                gsap.to(this.controls.target, { x: mesh.position.x, y: mesh.position.y, z: mesh.position.z, duration: 0.5 });
            },

            deselectActive() {
                if (this.activeBoxId) {
                    const mesh = this.meshes.find(m => m.userData.id === this.activeBoxId);
                    if (mesh) {
                        if (Array.isArray(mesh.material)) {
                            mesh.material.forEach(mat => mat.emissive.setHex(0x000000));
                        } else {
                            mesh.material.emissive.setHex(0x000000);
                        }
                    }
                    const uiItem = document.getElementById(`ui-item-${this.activeBoxId}`);
                    if (uiItem) uiItem.classList.remove('active');
                }
                this.activeBoxId = null;
                document.getElementById('interactive-panel').style.display = 'none';
            },

            toggleTargetingMode(forceState = null) {
                this.isTargetingMode = forceState !== null ? forceState : !this.isTargetingMode;
                const btn = document.getElementById('btn-target');
                if (this.isTargetingMode) {
                    btn.style.background = '#059669'; // Green active state
                    btn.innerHTML = '<span id="target-icon">📍</span> Quét... (Click Đích)';
                    this.container.style.cursor = 'crosshair';
                } else {
                    btn.style.background = 'rgba(0,0,0,0.5)';
                    btn.innerHTML = '<span id="target-icon">📍</span> Quét Bắt Điểm';
                    this.container.style.cursor = 'default';
                }
            },

            snapActiveToPoint(worldPoint) {
                const mesh = this.meshes.find(m => m.userData.id === this.activeBoxId);
                const data = this.boxesData.find(b => b.id === this.activeBoxId);
                if (!mesh || !data) return;

                const bbox = new THREE.Box3().setFromObject(mesh);
                const halfVisualHeight = (bbox.max.y - bbox.min.y) / 2;
                const pLen = this.inpPallet.value === "1.9" ? 1900 : 1200;
                
                const finalY = worldPoint.y + halfVisualHeight;
                mesh.position.set(worldPoint.x, finalY, worldPoint.z);
                mesh.updateMatrixWorld(true);
                
                const oh = this.checkOverhangInfo(mesh);
                if (oh.isFatal) {
                     const ok = confirm(`Cảnh báo: Kiện hàng thả lồi ra ngoài mép Pallet tới ${oh.amount.toFixed(0)}mm. Xác nhận đặt tràn mép?`);
                     if (!ok) {
                         const currentX = data.x - (pLen/2) + data.offsetX;
                         const currentY = data.z;
                         const currentZ = data.y - 500;
                         mesh.position.set(currentX, currentY, currentZ);
                         return; 
                     }
                }
                
                gsap.fromTo(mesh.position, {y: finalY + 400}, {x: worldPoint.x, y: finalY, z: worldPoint.z, duration: 0.3, ease: "power2.out"});
                
                data.x = worldPoint.x + (pLen/2) - data.offsetX;
                data.z = finalY;
                data.y = worldPoint.z + 500;
                
                this.updateStats();
                this.updateCartonUI(data);
            },

            addManualPallet() {
                const currentNum = this.palletMeshes ? this.palletMeshes.length : 0;
                const newNum = Math.max(1, currentNum) + 1;
                const pLen = this.inpPallet.value === "1.9" ? 1900 : 1200;
                this.rebuildNPallets(newNum, pLen);
                this.updateStats();
            },

            copyTopLayer() {
                const validBoxes = this.boxesData.filter(b => !b.deleted);
                if (validBoxes.length === 0) {
                    this.lblStatus.innerText = 'Chưa có kiện hàng nào để copy!';
                    this.lblStatus.style.color = '#ef4444';
                    return;
                }

                let maxZ = -Infinity;
                for (let b of validBoxes) {
                    if (b.z > maxZ) maxZ = b.z;
                }

                const topLayerBoxes = validBoxes.filter(b => Math.abs(b.z - maxZ) < 5);
                if (topLayerBoxes.length === 0) return;

                let addedCount = 0;
                topLayerBoxes.forEach(b => {
                    const newBox = Object.assign({}, b);
                    newBox.id = Date.now() + Math.random();
                    newBox.z = b.z + b.h; 
                    newBox.layer = (b.layer || 0) + 1;
                    newBox.isManual = true; 

                    this.boxesData.push(newBox);
                    this.drawNextCarton(true); 
                    addedCount++;
                });

                this.lblStatus.innerText = `Đã nhân bản ${addedCount} kiện hàng lên lớp tiếp theo.`;
                this.lblStatus.style.color = '#10b981';
            },

            copyBlock() {
                const validBoxes = this.boxesData.filter(b => !b.deleted);
                if (validBoxes.length === 0) {
                    this.lblStatus.innerText = 'Chưa có kiện hàng nào để copy!';
                    this.lblStatus.style.color = '#ef4444';
                    return;
                }

                let maxTop = 170; // Mặt sàn Pallet Base Z
                let maxLayer = 0;
                for (let b of validBoxes) {
                    const top = b.z + (b.h / 2);
                    if (top > maxTop) maxTop = top;
                    if (b.layer > maxLayer) maxLayer = b.layer;
                }

                // Tính toán chiều cao của toàn bộ khối để dịch chuyển lên trên
                const blockHeight = maxTop - 170;
                if (blockHeight <= 0) return;

                let addedCount = 0;
                validBoxes.forEach(b => {
                    const newBox = Object.assign({}, b);
                    newBox.id = Date.now() + Math.random();
                    newBox.z = b.z + blockHeight; 
                    newBox.layer = (b.layer || 0) + maxLayer + 1;
                    newBox.isManual = true; 

                    this.boxesData.push(newBox);
                    this.drawNextCarton(true);
                    addedCount++;
                });

                this.lblStatus.innerText = `Đã nhân bản toàn bộ khối (${addedCount} kiện) lên trên.`;
                this.lblStatus.style.color = '#10b981';

            checkCollisionWithGravity(mesh, dx, dy, dz) {
                const origX = mesh.position.x;
                const origY = mesh.position.y;
                const origZ = mesh.position.z;
                
                const origBBox = new THREE.Box3().setFromObject(mesh);
                const origBottom = origBBox.min.y;

                mesh.position.set(origX + dx, origY + dz, origZ + dy);
                mesh.updateMatrixWorld(true);
                
                let blocked = false;
                const newBBox = new THREE.Box3().setFromObject(mesh);
                newBBox.expandByScalar(-1); // tolerance
                
                if (newBBox.min.y < 170) blocked = true;
                
                let highestSupportY = 170; 
                
                for(let m of this.meshes) {
                    if (m === mesh || m.userData.deleted) continue;
                    
                    const mBBox = new THREE.Box3().setFromObject(m);
                    mBBox.expandByScalar(-1);
                    
                    if (newBBox.intersectsBox(mBBox)) {
                        blocked = true;
                    }
                    
                    if (dx !== 0 || dy !== 0) { // lateral
                         if (newBBox.max.x > mBBox.min.x && newBBox.min.x < mBBox.max.x &&
                             newBBox.max.z > mBBox.min.z && newBBox.min.z < mBBox.max.z) {
                               if (mBBox.max.y <= origBottom + 5) { 
                                   if (mBBox.max.y > highestSupportY) highestSupportY = mBBox.max.y;
                               }
                         }
                    }
                }
                
                mesh.position.set(origX, origY, origZ);
                mesh.updateMatrixWorld(true);
                
                return { blocked, highestSupportY };
            },

            checkOverhangInfo(mesh) {
                const bBox = new THREE.Box3().setFromObject(mesh);
                bBox.expandByScalar(-1);
                let bestMatch = null;
                
                for(let pMesh of this.palletMeshes) {
                     const pBox = new THREE.Box3().setFromObject(pMesh);
                     let overX = 0; let overZ = 0;
                     if (bBox.min.x < pBox.min.x) overX = Math.max(overX, pBox.min.x - bBox.min.x);
                     if (bBox.max.x > pBox.max.x) overX = Math.max(overX, bBox.max.x - pBox.max.x);
                     if (bBox.min.z < pBox.min.z) overZ = Math.max(overZ, pBox.min.z - bBox.min.z);
                     if (bBox.max.z > pBox.max.z) overZ = Math.max(overZ, bBox.max.z - pBox.max.z);
                     
                     const totalOverhang = Math.max(overX, overZ);
                     if (!bestMatch || totalOverhang < bestMatch.amount) {
                         const pLen = pBox.max.x - pBox.min.x;
                         const tolX = pLen * 0.05;
                         const tolZ = 1000 * 0.05;
                         bestMatch = { amount: totalOverhang, isFatal: (overX > tolX || overZ > tolZ) };
                     }
                }
                return bestMatch || { amount: 0, isFatal: false };
            },

            nudgeCarton(dx, dy, dz) {
                if (!this.activeBoxId) return;
                const mesh = this.meshes.find(m => m.userData.id === this.activeBoxId);
                const data = this.boxesData.find(b => b.id === this.activeBoxId);
                if (!mesh || !data) return;
                
                const pLen = this.inpPallet.value === "1.9" ? 1900 : 1200;
                gsap.killTweensOf(mesh.position);
                const trueX = data.x - (pLen/2) + data.offsetX;
                const trueY = data.z;
                const trueZ = data.y - 500;
                mesh.position.set(trueX, trueY, trueZ);
                mesh.updateMatrixWorld(true);
                
                let r = this.checkCollisionWithGravity(mesh, dx, dy, dz);
                let wasRotatedToFit = false;
                
                if (r.blocked) {
                    const origRot = mesh.rotation.y;
                    mesh.rotation.y = origRot === 0 ? (Math.PI / 2) : 0;
                    mesh.updateMatrixWorld(true);
                    
                    let rRot = this.checkCollisionWithGravity(mesh, dx, dy, dz);
                    if (!rRot.blocked) {
                        data.angle = data.angle === 0 ? 90 : 0;
                        r = rRot;
                        wasRotatedToFit = true;
                    } else {
                        mesh.rotation.y = origRot;
                        mesh.updateMatrixWorld(true);
                    }
                }
                
                if (r.blocked) {
                    gsap.to(mesh.position, {
                        x: trueX + (dx>0?5:(dx<0?-5:0)), 
                        y: trueY + (dz>0?5:(dz<0?-5:0)),
                        z: trueZ + (dy>0?5:(dy<0?-5:0)),
                        yoyo: true, repeat: 1, duration: 0.05 
                    });
                    return;
                }
                
                let newX = trueX + dx;
                let newZ = trueZ + dy;
                let newY = trueY + dz;
                
                if (dx !== 0 || dy !== 0) {
                    const bbox = new THREE.Box3().setFromObject(mesh);
                    const halfY = (bbox.max.y - bbox.min.y) / 2;
                    const dropY = r.highestSupportY + halfY + 0.1; 
                    if (newY > dropY) newY = dropY;
                }
                
                mesh.position.set(newX, newY, newZ);
                mesh.updateMatrixWorld(true);
                const oh = this.checkOverhangInfo(mesh);
                if (oh.isFatal) {
                     const ok = confirm(`Cảnh báo: Kiện hàng lồi ra ngoài mép Pallet tới ${oh.amount.toFixed(0)}mm (Vượt trần 5%). Đặt ép lấn ranh?`);
                     if (!ok) {
                          mesh.position.set(trueX, trueY, trueZ);
                          if (wasRotatedToFit) {
                              mesh.rotation.y = data.angle === 90 ? 0 : (Math.PI/2);
                              data.angle = data.angle === 90 ? 0 : 90;
                          }
                          return;
                     } // else keep going
                }
                
                mesh.position.set(wasRotatedToFit ? trueX : mesh.position.x, trueY, trueZ);
                gsap.to(mesh.position, { x: newX, y: newY, z: newZ, duration: 0.2, ease:"power1.out" });
                
                data.x = newX + (pLen/2) - data.offsetX;
                data.z = newY;
                data.y = newZ + 500;
                
                this.updateStats();
                this.updateCartonUI(data);
            },

            rotateActive() {
                if (!this.activeBoxId) return;
                const mesh = this.meshes.find(m => m.userData.id === this.activeBoxId);
                const data = this.boxesData.find(b => b.id === this.activeBoxId);
                if (!mesh || !data) return;
                
                data.angle = data.angle === 0 ? 90 : 0;
                gsap.to(mesh.rotation, { y: mesh.rotation.y + (Math.PI / 2), duration: 0.3, ease: "back.out(1.5)" });
                this.lblStatus.innerText = `Đã xoay thùng đang chọn sang ${data.angle}°. Nhấn R để đổi lại.`;
                this.updateStats();
                this.updateCartonUI(data);
            },

            flipActive() {
                if (!this.activeBoxId) return;
                const mesh = this.meshes.find(m => m.userData.id === this.activeBoxId);
                const data = this.boxesData.find(b => b.id === this.activeBoxId);
                if (!mesh || !data) return;
                
                // Cập nhật đổi W ↔ H (chiều cao h mới)
                const oldH = data.h;
                data.h = data.w;
                data.w = oldH;
                
                // Cân đối lại tọa độ y (để khối không bị lún hoặc lơ lửng)
                const bottomZ = data.z - oldH/2;
                data.z = bottomZ + data.h/2;
                
                gsap.to(mesh.rotation, { x: mesh.rotation.x + (Math.PI / 2), duration: 0.3, ease: "back.out(1.5)" });
                gsap.to(mesh.position, { y: data.z, duration: 0.3, ease: "back.out(1.5)" });
                this.updateStats();
                this.updateCartonUI(data);
            },

            rollActive() {
                if (!this.activeBoxId) return;
                const mesh = this.meshes.find(m => m.userData.id === this.activeBoxId);
                const data = this.boxesData.find(b => b.id === this.activeBoxId);
                if (!mesh || !data) return;
                
                // Cập nhật đổi L ↔ H
                const oldH = data.h;
                data.h = data.l;
                data.l = oldH;
                
                const bottomZ = data.z - oldH/2;
                data.z = bottomZ + data.h/2;
                
                gsap.to(mesh.rotation, { z: mesh.rotation.z + (Math.PI / 2), duration: 0.3, ease: "back.out(1.5)" });
                gsap.to(mesh.position, { y: data.z, duration: 0.3, ease: "back.out(1.5)" });
                this.updateStats();
                this.updateCartonUI(data);
            },

            removeActive() {
                if (!this.activeBoxId) return;
                const targetId = this.activeBoxId;
                const meshIdx = this.meshes.findIndex(m => m.userData.id === targetId);
                const dataIdx = this.boxesData.findIndex(b => b.id === targetId);
                
                if (meshIdx !== -1) {
                    const mesh = this.meshes[meshIdx];
                    this.scene.remove(mesh);
                    mesh.geometry.dispose(); 
                    if (Array.isArray(mesh.material)) {
                        mesh.material.forEach(mat => { if(mat.map) mat.map.dispose(); mat.dispose(); });
                    } else {
                        if (mesh.material.map) mesh.material.map.dispose();
                        mesh.material.dispose();
                    }
                    this.meshes.splice(meshIdx, 1);
                }
                if (dataIdx !== -1) {
                    this.boxesData[dataIdx].deleted = true; // Mark deleted to remove from stats
                    
                    // Loại bỏ thẻ UI trên thanh danh sách
                    const uiItem = document.getElementById(`ui-item-${targetId}`);
                    if (uiItem) uiItem.remove();
                }
                
                this.deselectActive();
                this.updateStats();
            },

            rebuildNPallets(numPallets, forcedLength) {
                if (this.palletMeshes) { this.palletMeshes.forEach(m => { this.scene.remove(m); m.geometry.dispose(); }); }
                this.palletMeshes = [];
                
                const palletWood = new THREE.MeshStandardMaterial({ color: '#8b5a2b', roughness: 0.9, metalness: 0.1 });
                const palletGeo = new THREE.BoxGeometry(forcedLength, 170, 1000);
                const gap = forcedLength === 1900 ? 2200 : 1500; 

                for (let i = 0; i < numPallets; i++) {
                    const mesh = new THREE.Mesh(palletGeo, palletWood);
                    mesh.position.set(i * gap, 85, 0); 
                    mesh.receiveShadow = true; mesh.castShadow = true;
                    this.scene.add(mesh);
                    this.palletMeshes.push(mesh);
                }
                
                if (numPallets > 1) {
                    const centerPoint = ((numPallets - 1) * gap) / 2;
                    gsap.to(this.controls.target, { x: centerPoint, duration: 1 });
                    gsap.to(this.camera.position, { x: centerPoint, z: 4000 + (numPallets * 500), duration: 1 });
                } else {
                    gsap.to(this.controls.target, { x: 0, duration: 1 });
                }
            },

            runAutoPlan() {
                if (this.orderList.length === 0) {
                    this.lblStatus.innerText = 'Vui lòng thêm order trước khi quy hoạch!'; return;
                }

                this.resetView(); 
                if (this.isPlaying) this.togglePlay();

                const pType = this.inpPallet.value;
                const pLen = pType === "1.9" ? 1900 : 1200;
                const allowMix = this.inpMix.value === "true";

                const resultPallets = this.palletizer.planMultipleOrders(this.orderList, allowMix, pType);
                this.rebuildNPallets(Math.max(1, resultPallets.length), pLen);

                this.boxesData = [];
                const gap = pLen === 1900 ? 2200 : 1500;

                resultPallets.forEach((palletObj, pIdx) => {
                    const offsetX = pIdx * gap;
                    palletObj.boxes.forEach(b => {
                        b.pIdx = pIdx;
                        b.offsetX = offsetX;
                        this.boxesData.push(b);
                    });
                });

                document.getElementById('btn-play').disabled = false;
                this.lblStatus.innerText = `Đã thiết kế xong: ${resultPallets.length} Pallet (${this.boxesData.length} kiện). Đang mô phỏng...`;
                this.togglePlay(true);
            },

                const data = localStorage.getItem('palletSimulatorState');
                if (!data) return false;

                    const state = JSON.parse(data);
                    if (!state || !state.boxes || state.boxes.length === 0) return false;

                        this.drawNextCarton(true);

                    this.isLoadingState = false;
                    return false;
                }

            onMouseDown(e) {
                if (e.button !== 0 || this.isTargetingMode || this.isDropMode) return;
                this.hasDragged = false;
                const rect = this.container.getBoundingClientRect();
                this.mouse.x = ((e.clientX - rect.left) / rect.width) * 2 - 1;
                this.mouse.y = -((e.clientY - rect.top) / rect.height) * 2 + 1;
                this.raycaster.setFromCamera(this.mouse, this.camera);
                
                const intersects = this.raycaster.intersectObjects(this.meshes);
                if (intersects.length > 0) {
                    this.controls.enabled = false;
                    this.isDragging = true;
                    const hitMesh = intersects[0].object;
                    
                    if (hitMesh.userData.id) {
                        this.activateCarton(hitMesh.userData.id);
                        this.dragPlane.setFromNormalAndCoplanarPoint(new THREE.Vector3(0, 1, 0), hitMesh.position);
                        if (this.raycaster.ray.intersectPlane(this.dragPlane, this.intersectionPoint)) {
                            this.dragOffset.copy(hitMesh.position).sub(this.intersectionPoint);
                        }
                        
                        // Hiển thị Ghost Box bằng kích thước và góc xoay của hộp đang chọn
                        if (this.ghostBox) {
                            this.ghostBox.geometry.dispose();

                            this.ghostBox.position.copy(hitMesh.position);
                            this.ghostBox.rotation.copy(hitMesh.rotation);
                            this.ghostBox.visible = true;
                        }
                    }
                }
            },
            
            calculateMagneticSnap(mesh, targetX, targetZ) {
                const threshold = 30; // Khoảng cách hút nam châm
                let bestX = targetX;
                let bestZ = targetZ;
                let minDiffX = threshold;
                let minDiffZ = threshold;

                const bbox = new THREE.Box3().setFromObject(mesh);
                const width = bbox.max.x - bbox.min.x;
                const depth = bbox.max.z - bbox.min.z;
                
                const pLen = this.inpPallet.value === "1.9" ? 1900 : 1200;
                const pW = 1000;

                const checkSnapX = (targetEdge, refEdge, isLeftEdge) => {
                    const diff = Math.abs(targetEdge - refEdge);
                    if (diff < minDiffX) {
                        minDiffX = diff;
                        bestX = isLeftEdge ? refEdge + width/2 : refEdge - width/2;
                    }
                };

                const checkSnapZ = (targetEdge, refEdge, isTopEdge) => {
                    const diff = Math.abs(targetEdge - refEdge);
                    if (diff < minDiffZ) {
                        minDiffZ = diff;
                        bestZ = isTopEdge ? refEdge + depth/2 : refEdge - depth/2;
                    }
                };

                // Snap với biên Pallet
                checkSnapX(targetX - width/2, -pLen/2, true);
                checkSnapX(targetX + width/2, pLen/2, false);
                checkSnapZ(targetZ - depth/2, -pW/2, true);
                checkSnapZ(targetZ + depth/2, pW/2, false);

                for (let m of this.meshes) {
                    if (m === mesh || m.userData.deleted) continue;

                    const mBBox = new THREE.Box3().setFromObject(m);
                    
                    // Giao nhau trục Z thì hút trục X
                    const zOverlap = (targetZ - depth/2 < mBBox.max.z + threshold) && (targetZ + depth/2 > mBBox.min.z - threshold);
                    if (zOverlap) {
                        checkSnapX(targetX + width/2, mBBox.min.x, false); // Phải target -> Trái m
                        checkSnapX(targetX - width/2, mBBox.max.x, true);  // Trái target -> Phải m
                        checkSnapX(targetX - width/2, mBBox.min.x, true);  // Trái target -> Trái m
                        checkSnapX(targetX + width/2, mBBox.max.x, false); // Phải target -> Phải m
                    }

                    // Giao nhau trục X thì hút trục Z
                    const xOverlap = (bestX - width/2 < mBBox.max.x + threshold) && (bestX + width/2 > mBBox.min.x - threshold);
                    if (xOverlap) {
                        checkSnapZ(targetZ + depth/2, mBBox.min.z, false); // Dưới target -> Trên m
                        checkSnapZ(targetZ - depth/2, mBBox.max.z, true);  // Trên target -> Dưới m
                        checkSnapZ(targetZ - depth/2, mBBox.min.z, true);  // Trên target -> Trên m
                        checkSnapZ(targetZ + depth/2, mBBox.max.z, false); // Dưới target -> Dưới m
                    }
                }

                return { snappedX: bestX, snappedZ: bestZ };
            },

            onMouseMove(e) {
                const rect = this.container.getBoundingClientRect();
                this.mouse.x = ((e.clientX - rect.left) / rect.width) * 2 - 1;
                this.mouse.y = -((e.clientY - rect.top) / rect.height) * 2 + 1;

                if (this.isDropMode && this.ghostBox) {
                    this.raycaster.setFromCamera(this.mouse, this.camera);
                    const dropPlane = new THREE.Plane(new THREE.Vector3(0, 1, 0), -170); // Mặt sàn Pallet
                    if (this.raycaster.ray.intersectPlane(dropPlane, this.intersectionPoint)) {
                        const snap = 10;
                        let snappedX = Math.round(this.intersectionPoint.x / snap) * snap;
                        let snappedZ = Math.round(this.intersectionPoint.z / snap) * snap;
                        
                        let h = parseFloat(this.inpH.value) || 200;
                        let l = parseFloat(this.inpL.value) || 400; 
                        let w = parseFloat(this.inpW.value) || 300;
                        const angle = Number.isFinite(this.dropPreviewAngle) ? this.dropPreviewAngle : parseInt(document.getElementById('inp-orient')?.value || 0);
                        const quarterIdx = Math.abs(Math.round(angle / 90)) % 2; // 0 for 0/180, 1 for 90/270
                        // Lật úp theo tùy chọn
                        if (this.dropPreviewFlip === 'flipW') [w, h] = [h, w];
                        else if (this.dropPreviewFlip === 'flipL') [l, h] = [h, l];
                        const effL = quarterIdx === 0 ? l : w;
                        const effW = quarterIdx === 0 ? w : l;

                        // Cập nhật kích thước & hiển thị Ghost Box cho chế độ thả tay
                        if (!this.ghostBox.userData) this.ghostBox.userData = {};
                        if (this.ghostBox.userData.effL !== effL || this.ghostBox.userData.effW !== effW || this.ghostBox.userData.h !== h) {
                            this.ghostBox.geometry.dispose();
                            this.ghostBox.geometry = new THREE.BoxGeometry(effL, h, effW);
                            this.ghostBox.userData.effL = effL;
                            this.ghostBox.userData.effW = effW;
                            this.ghostBox.userData.h = h;
                        }
                        this.ghostBox.rotation.set(0, THREE.MathUtils.degToRad(angle % 360), 0);
                        this.ghostBox.visible = true;
                        this.updateGhostInfo({
                            angle,
                            l,
                            w,
                            h,
                            overhang: false,
                            flip: this.dropPreviewFlip
                        });
                        
                        let highestSupportY = 170;
                        const tMinX = snappedX - effL/2 + 1;
                        const tMaxX = snappedX + effL/2 - 1;
                        const tMinZ = snappedZ - effW/2 + 1;
                        const tMaxZ = snappedZ + effW/2 - 1;
                        
                        for (let m of this.meshes) {
                            if (m.userData.deleted) continue;
                            const mBBox = new THREE.Box3().setFromObject(m);
                            mBBox.expandByScalar(-1);
                            if (tMinX < mBBox.max.x && tMaxX > mBBox.min.x && tMinZ < mBBox.max.z && tMaxZ > mBBox.min.z) {
                                if (mBBox.max.y > highestSupportY) highestSupportY = mBBox.max.y;
                            }
                        }
                        
                        const predictedY = highestSupportY + (h / 2);
                        this.ghostBox.position.set(snappedX, predictedY, snappedZ);
                        
                        const pLen = this.inpPallet.value === "1.9" ? 1900 : 1200;
                        const isOverhang = (tMinX - 1 < -pLen/2 || tMaxX + 1 > pLen/2 || tMinZ - 1 < -500 || tMaxZ + 1 > 500);
                        this.ghostBox.material.color.setHex(isOverhang ? 0xef4444 : 0x10b981);
                this.updateGhostInfo({ angle, l, w, h, overhang: isOverhang, flip: this.dropPreviewFlip });
                    }
                    return;
                }

                if (!this.isDragging || !this.activeBoxId) return;
                this.hasDragged = true;
                this.raycaster.setFromCamera(this.mouse, this.camera);
                
                if (this.raycaster.ray.intersectPlane(this.dragPlane, this.intersectionPoint)) {
                    const newPos = this.intersectionPoint.add(this.dragOffset);
                    const mesh = this.meshes.find(m => m.userData.id === this.activeBoxId);
                    const data = this.boxesData.find(b => b.id === this.activeBoxId);
                    if (mesh && data) {
                        // Snap grid lưới 10mm
                        const snap = 10;
                        let snappedX = Math.round(newPos.x / snap) * snap;
                        let snappedZ = Math.round(newPos.z / snap) * snap;
                        
                        // Hút nam châm (Auto-Snap)
                        const magneticSnap = this.calculateMagneticSnap(mesh, snappedX, snappedZ);
                        snappedX = magneticSnap.snappedX;
                        snappedZ = magneticSnap.snappedZ;
                        
                        // Di chuyển khối thật mượt mà theo con trỏ chuột
                        mesh.position.x = newPos.x;
                        mesh.position.z = newPos.z;
                        
                        // Tính toán độ cao rơi (Gravity Snap) cho Ghost Box
                        let predictedY = mesh.position.y;
                        let highestSupportY = 170; // Mặt sàn Pallet (Base Z)

                        const width = data.angle === 0 ? data.l : data.w;
                        const depth = data.angle === 0 ? data.w : data.l;
                        
                        const tMinX = snappedX - width/2 + 1;
                        const tMaxX = snappedX + width/2 - 1;
                        const tMinZ = snappedZ - depth/2 + 1;
                        const tMaxZ = snappedZ + depth/2 - 1;

                        for (let m of this.meshes) {
                            if (m === mesh || m.userData.deleted) continue;
                            const mBBox = new THREE.Box3().setFromObject(m);
                            mBBox.expandByScalar(-1); // Dung sai
                            
                            if (tMinX < mBBox.max.x && tMaxX > mBBox.min.x && tMinZ < mBBox.max.z && tMaxZ > mBBox.min.z) {
                                if (mBBox.max.y > highestSupportY) highestSupportY = mBBox.max.y;
                            }
                        }
                        
                        predictedY = highestSupportY + (data.h / 2);

                        // Cập nhật vị trí bóng mờ (Ghost Box) kèm tọa độ Y đã dự đoán
                        if (this.ghostBox) {
                            this.ghostBox.position.set(snappedX, predictedY, snappedZ);
                            
                            // Cảnh báo màu Đỏ nếu bóng mờ bị lồi ra khỏi biên Pallet
                            const pLen = this.inpPallet.value === "1.9" ? 1900 : 1200;
                            const pWid = 1000;
                            const palletMinX = (data.offsetX || 0) - pLen/2;
                            const palletMaxX = (data.offsetX || 0) + pLen/2;
                            const palletMinZ = -pWid/2;
                            const palletMaxZ = pWid/2;
                            
                            const isOverhang = (tMinX - 1 < palletMinX || tMaxX + 1 > palletMaxX || tMinZ - 1 < palletMinZ || tMaxZ + 1 > palletMaxZ);
                            
                            this.ghostBox.material.color.setHex(isOverhang ? 0xef4444 : 0x10b981);
                        }
                    }
                }
            },
            
            onMouseUp(e) {
                if (this.isDragging) {
                    this.isDragging = false;
                    this.controls.enabled = true; // Bật lại OrbitCamera
                    
                    if (this.ghostBox) {
                        this.ghostBox.visible = false;
                    }
                    
                    const mesh = this.meshes.find(m => m.userData.id === this.activeBoxId);
                    const data = this.boxesData.find(b => b.id === this.activeBoxId);
                    
                    if (mesh && data && this.ghostBox) {
                        const targetX = this.ghostBox.position.x;
                        const targetY = this.ghostBox.position.y;
                        const targetZ = this.ghostBox.position.z;
                        
                        // Thả rớt (Gravity Drop)
                        mesh.position.set(targetX, mesh.position.y, targetZ);
                        gsap.to(mesh.position, { y: targetY, duration: 0.2, ease: "power2.out" });
                        
                        const pLen = this.inpPallet.value === "1.9" ? 1900 : 1200;
                        data.x = targetX + (pLen/2) - data.offsetX;
                        data.z = targetY;
                        data.y = targetZ + 500;
                        
                        this.updateStats();
                        this.updateCartonUI(data);
                    }
                }
            },
            
            onKeyDown(e) {
                // Không xử lý phím khi đang nhập form
                if (document.activeElement.tagName === 'INPUT' || document.activeElement.tagName === 'SELECT') return;

                // Cho phép xoay trước khi đặt pallet ở chế độ thả tay (ghost box)
                if (this.isDropMode && !this.activeBoxId) {
                    const key = e.key.toLowerCase();
                    if (key === 'r' || key === 'q' || key === 'e') {
                        const delta = key === 'q' ? -90 : (key === 'e' ? 90 : 90);
                        this.dropPreviewAngle = ((this.dropPreviewAngle + delta) % 360 + 360) % 360; // chuẩn hóa 0-359
                        const orientSel = document.getElementById('inp-orient');
                        if (orientSel) orientSel.value = (Math.abs(Math.round(this.dropPreviewAngle / 90)) % 2 === 0) ? 0 : 90;
                        this.updateGhostInfo({
                            angle: this.dropPreviewAngle,
                            l: parseFloat(this.inpL.value) || 400,
                            w: parseFloat(this.inpW.value) || 300,
                            h: parseFloat(this.inpH.value) || 200,
                            overhang: false,
                            flip: this.dropPreviewFlip
                        });
                        this.lblStatus.innerText = `Đang xoay hướng đặt thủ công: ${this.dropPreviewAngle}° (Q/E/R để đổi)`;
                        return;
                    } else if (key === 'f' || key === 't') {
                        // F: lật ngang (W -> H), T: lật dài (L -> H)
                        this.dropPreviewFlip = key === 'f'
                            ? (this.dropPreviewFlip === 'flipW' ? 'none' : 'flipW')
                            : (this.dropPreviewFlip === 'flipL' ? 'none' : 'flipL');
                        this.updateGhostInfo({
                            angle: this.dropPreviewAngle,
                            l: parseFloat(this.inpL.value) || 400,
                            w: parseFloat(this.inpW.value) || 300,
                            h: parseFloat(this.inpH.value) || 200,
                            overhang: false,
                            flip: this.dropPreviewFlip
                        });
                        this.lblStatus.innerText = this.dropPreviewFlip === 'none'
                            ? 'Đã tắt chế độ lật. Chồng theo layer hiện tại với kích thước chuẩn.'
                            : (this.dropPreviewFlip === 'flipW' ? 'Đang lật ngang: W trở thành H.' : 'Đang lật chiều dài: L trở thành H.');
                        return;
                    }
                }

                if (!this.activeBoxId) return;

                switch(e.key.toLowerCase()) {
                    case 'r': this.rotateActive(); break;
                    case 'f': this.flipActive(); break;
                    case 't': this.rollActive(); break;
                    case 'delete': 
                    case 'backspace': 
                        this.removeActive(); break;
                    case 'arrowleft': this.nudgeCarton(-50, 0, 0); break;
                    case 'arrowright': this.nudgeCarton(50, 0, 0); break;
                    case 'arrowup': this.nudgeCarton(0, -50, 0); break;
                    case 'arrowdown': this.nudgeCarton(0, 50, 0); break;
                }
            },

            resetView() {
                this.deselectActive();
                this.meshes.forEach(m => { this.scene.remove(m); m.geometry.dispose(); m.material.dispose(); });
                this.meshes = [];
                this.totalCartonsLoaded = 0;
                document.getElementById('carton-list').innerHTML = '';
                
                if (this.layerLabelsGroup) {
                    while(this.layerLabelsGroup.children.length > 0) {
                        const child = this.layerLabelsGroup.children[0];
                        if(child.material.map) child.material.map.dispose();
                        child.material.dispose();
                        this.layerLabelsGroup.remove(child);
                    }
                }
            },
            
            togglePlay(forcePlay=null) {
                if (forcePlay !== null) this.isPlaying = !forcePlay; 
                const validBoxes = this.boxesData.filter(b => !b.deleted);
                if (this.totalCartonsLoaded >= validBoxes.length && !this.isPlaying) return;
                
                this.isPlaying = !this.isPlaying;
                const btn = document.getElementById('btn-play');
                const txt = document.getElementById('play-text');
                const icn = document.getElementById('play-icon');
                
                if (this.isPlaying) {
                    icn.innerText = '⏸'; txt.innerText = 'Pause'; btn.style.background = '#f59e0b';
                    this.playInterval = setInterval(() => {
                        if (this.totalCartonsLoaded >= validBoxes.length) {
                            this.togglePlay(); 
                            if (this.lblStatus) this.lblStatus.innerText = '✅ Hoàn tất mô phỏng Palletizing!';
                        } else {
                            this.drawNextCarton();
                        }
                    }, 40); 
                } else {
                    icn.innerText = '▶'; txt.innerText = 'Resume'; btn.style.background = 'var(--secondary)';
                    clearInterval(this.playInterval);
                }
            },
            
            createLabelTexture(text, bgColorHex) {
                const canvas = document.createElement('canvas');
                canvas.width = 256;
                canvas.height = 256;
                const ctx = canvas.getContext('2d');

                // Tô màu nền cho thùng
                ctx.fillStyle = bgColorHex;
                ctx.fillRect(0, 0, canvas.width, canvas.height);

                // Vẽ viền nhạt để thùng nhìn chi tiết hơn
                ctx.strokeStyle = 'rgba(255, 255, 255, 0.4)';
                ctx.lineWidth = 8;
                ctx.strokeRect(8, 8, canvas.width - 16, canvas.height - 16);

                // Vẽ số thứ tự ở chính giữa
                ctx.fillStyle = '#ffffff';
                ctx.font = 'bold 100px sans-serif';
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                ctx.fillText(text, canvas.width / 2, canvas.height / 2);

                return new THREE.CanvasTexture(canvas);
            },

            createLayerLabelSprite(message, colorHex) {
                const canvas = document.createElement('canvas');
                const context = canvas.getContext('2d');
                const fontSize = 28;
                context.font = `bold ${fontSize}px sans-serif`;
                const textWidth = context.measureText(message).width;
                
                canvas.width = textWidth + 32;
                canvas.height = fontSize + 24;
                
                // Vẽ nền
                context.fillStyle = 'rgba(17, 24, 39, 0.85)';
                context.fillRect(0, 0, canvas.width, canvas.height);
                
                // Vẽ viền
                context.strokeStyle = colorHex;
                context.lineWidth = 3;
                context.strokeRect(1.5, 1.5, canvas.width - 3, canvas.height - 3);
                
                // Vẽ chữ
                context.fillStyle = '#ffffff';
                context.textAlign = 'center';
                context.textBaseline = 'middle';
                context.fillText(message, canvas.width / 2, canvas.height / 2 + 2);
                
                const texture = new THREE.CanvasTexture(canvas);
                const spriteMaterial = new THREE.SpriteMaterial({ map: texture, depthTest: false }); 
                const sprite = new THREE.Sprite(spriteMaterial);
                sprite.renderOrder = 999; // Ưu tiên hiển thị đè lên trên khối
                
                const scale = 1.8;
                sprite.scale.set(canvas.width * scale, canvas.height * scale, 1);
                
                return sprite;
            },

            updateLayerLabels() {
                if (!this.layerLabelsGroup) return;
                while(this.layerLabelsGroup.children.length > 0) {
                    const child = this.layerLabelsGroup.children[0];
                    if(child.material.map) child.material.map.dispose();
                    child.material.dispose();
                    this.layerLabelsGroup.remove(child);
                }

                const activeBoxes = this.boxesData.filter((b, i) => !b.deleted && (i < this.totalCartonsLoaded || b.isManual));
                if (activeBoxes.length === 0) return;

                const palletsMap = new Map();
                activeBoxes.forEach(b => {
                    if (!palletsMap.has(b.pIdx)) palletsMap.set(b.pIdx, []);
                    palletsMap.get(b.pIdx).push(b);
                });

                const pLen = this.inpPallet.value === "1.9" ? 1900 : 1200;

                palletsMap.forEach((boxes, pIdx) => {
                    const layers = [];
                    boxes.forEach(b => {
                        const topZ = b.z + b.h / 2;
                        // Gom nhóm các hộp có cùng mặt phẳng trần (Dung sai 10mm)
                        let foundLayer = layers.find(l => Math.abs(l.topZ - topZ) < 10);
                        if (foundLayer) {
                            foundLayer.boxes.push(b);
                            // Cập nhật lại trung bình độ cao mặt phẳng
                            foundLayer.topZ = (foundLayer.topZ * (foundLayer.boxes.length - 1) + topZ) / foundLayer.boxes.length;
                        } else {
                            layers.push({ topZ: topZ, boxes: [b] });
                        }
                    });

                    layers.sort((a, b) => a.topZ - b.topZ);

                    layers.forEach((layer, idx) => {
                        const count = layer.boxes.length;
                        const message = `Lớp ${idx + 1}: ${count} hộp`;
                        const sprite = this.createLayerLabelSprite(message, '#10b981');
                        
                        let sumX = 0, sumZ = 0;
                        layer.boxes.forEach(b => {
                            sumX += (b.x - pLen/2 + b.offsetX);
                            sumZ += (b.y - 500);
                        });
                        // Tính toán tâm của lớp hàng để treo nhãn
                        const avgX = sumX / count;
                        const avgZ = sumZ / count;
                        
                        sprite.position.set(avgX, layer.topZ + 60, avgZ); // Lơ lửng cách mặt lớp 60mm
                        this.layerLabelsGroup.add(sprite);
                    });
                });
            },

            drawNextCarton(isManual = false) {
                const idx = isManual ? (this.boxesData.length - 1) : this.totalCartonsLoaded;
                const data = this.boxesData[idx];
                if (!data || data.deleted) return;
                
                const w = data.angle === 0 ? data.l : data.w;
                const d = data.angle === 0 ? data.w : data.l;
                const geo = new THREE.BoxGeometry(w - 2, data.h - 2, d - 2); 
                
                if (!this.skuColorMap[data.sku]) {
                    const hue = Math.floor(Math.random() * 360);
                    const c = new THREE.Color();
                    c.setHSL(hue/360, 0.8, 0.55);
                    this.skuColorMap[data.sku] = '#' + c.getHexString();
                }
                const hexColor = this.skuColorMap[data.sku];
                
                // Tạo chất liệu (Material) có chứa texture dán nhãn số
                const boxNum = idx + 1;
                const tex = this.createLabelTexture(boxNum.toString(), hexColor);
                const matTex = new THREE.MeshStandardMaterial({ color: 0xffffff, map: tex, roughness: 0.6, metalness: 0.2 });
                const matPlain = new THREE.MeshStandardMaterial({ color: hexColor, roughness: 0.6, metalness: 0.2 });
                const materials = [matTex, matTex, matTex, matPlain, matTex, matTex]; // Dán nhãn lên tất cả trừ mặt đáy (bottom)
                
                const mesh = new THREE.Mesh(geo, materials);
                mesh.add(new THREE.LineSegments(new THREE.EdgesGeometry(geo), new THREE.LineBasicMaterial({ color: 0xffffff, transparent: true, opacity: 0.15 })));
                mesh.castShadow = true; mesh.receiveShadow = true;
                
                const pLen = this.inpPallet.value === "1.9" ? 1900 : 1200;
                
                // Trục tọa độ Threejs: X (trái/phải), Y (lên/xuống = Z của thùng), Z (chiều sâu = Y của thùng)
                mesh.position.set(data.x - (pLen/2) + data.offsetX, data.z, data.y - 500);
                mesh.rotation.y = data.angle === 90 ? (Math.PI / 2) : 0;
                
                mesh.userData = { id: data.id }; // Attach ID for raycasting
                
                this.scene.add(mesh);
                this.meshes.push(mesh);
                
                gsap.from(mesh.position, { y: data.z + 800, duration: 0.5, ease: "power2.out" });
                if (!this.isLoadingState) {
                    gsap.from(mesh.position, { y: data.z + 800, duration: 0.5, ease: "power2.out" });
                }

                this.addToListUI(data, hexColor);
                if (!isManual) this.totalCartonsLoaded++;
                this.updateStats();
            },

                        <div>
                            <span class="carton-layer" style="color: ${colorHex}; border: 1px solid ${colorHex}55">${data.isManual ? 'MANUAL' : 'L' + data.layer}</span>
                            <span style="color: #ef4444; margin-left: 8px; cursor: pointer; padding: 2px;" onclick="event.stopPropagation(); app.activateCarton(${data.id}); app.removeActive();" title="Xóa thùng này">✖</span>
                        </div>
                    </div>
                    <div class="carton-coord" style="font-size: 10px; color: var(--text-muted); margin-top: 2px;">
                        Tọa độ (X, Y, Z): ${Math.round(data.x)}, ${Math.round(data.y)}, ${Math.round(data.z)}
                    </div>
                `;
                div.onclick = () => this.activateCarton(data.id);
                document.getElementById('carton-list').prepend(div);
            },

                    if (coordDiv) {
                        coordDiv.innerText = `Tọa độ (X, Y, Z): ${Math.round(data.x)}, ${Math.round(data.y)}, ${Math.round(data.z)}`;
                    }
                }
            },
            
            updateStats() {
                if (this.isLoadingState) return; // Không cần cập nhật UI nhiều lần khi đang Restore
                let currentCount = 0;
                let currentWeight = 0;
                let totalCBM = 0;
                let highestPalletIndex = -1;
                let maxZ = 170; // Sàn gỗ cơ bản

                let minX = Infinity, maxX = -Infinity;
                let minY = Infinity, maxY = -Infinity;
                let maxZ_P0 = 170;
                let hasCartons = false;
                
                for(let i = 0; i < this.boxesData.length; i++) {
                    const b = this.boxesData[i];
                    // Count if not deleted and (either loaded via auto-play or is manual)
                    if (b && !b.deleted && (i < this.totalCartonsLoaded || b.isManual)) {
                        currentCount++;
                        currentWeight += b.weight || 0;
                        totalCBM += (b.cbm || 0);
                        if (b.pIdx > highestPalletIndex) highestPalletIndex = b.pIdx;
                        
                        let topOfBox = b.z + (b.h / 2);
                        if (topOfBox > maxZ) maxZ = topOfBox;

                        if (b.pIdx === 0) {
                            hasCartons = true;
                            const effL = b.angle === 0 ? b.l : b.w;
                            const effW = b.angle === 0 ? b.w : b.l;
                            
                            const bMinX = b.x - effL/2;
                            const bMaxX = b.x + effL/2;
                            const bMinY = b.y - effW/2;
                            const bMaxY = b.y + effW/2;
                            
                            if (bMinX < minX) minX = bMinX;
                            if (bMaxX > maxX) maxX = bMaxX;
                            if (bMinY < minY) minY = bMinY;
                            if (bMaxY > maxY) maxY = bMaxY;
                            if (topOfBox > maxZ_P0) maxZ_P0 = topOfBox;
                        }
                    }
                }
                
                const displayPallets = Math.max(this.palletMeshes ? this.palletMeshes.length : 1, highestPalletIndex + 1);
                document.getElementById('lbl-pallets').innerText = displayPallets;
                document.getElementById('lbl-loaded').innerText = currentCount;
                document.getElementById('lbl-total-cbm').innerText = totalCBM.toFixed(3);
                document.getElementById('lbl-total-weight').innerText = (currentWeight / 1000).toFixed(2);
                
                const zLabel = document.getElementById('lbl-max-z');
                zLabel.innerText = Math.floor(maxZ);
                if (maxZ > 2600) { zLabel.style.color = '#ef4444'; } else { zLabel.style.color = 'white'; }

                // --- Update Spatial Dimensions UI ---
                const pLen = this.inpPallet.value === "1.9" ? 1900 : 1200;
                const pWid = 1000;
                const pMaxH = 2600;

                let usedL = 0, usedW = 0;
                let isOverhang = false;
                if (hasCartons) {
                    usedL = maxX - minX;
                    usedW = maxY - minY;
                    // Check nếu tọa độ các mép hộp tràn ra khỏi giới hạn Pallet (Dung sai 1mm)
                    if (minX < -1 || maxX > pLen + 1 || minY < -1 || maxY > pWid + 1) {
                        isOverhang = true;
                    }
                }

                const freeL = Math.max(0, pLen - usedL);
                const freeW = Math.max(0, pWid - usedW);
                const freeH = Math.max(0, pMaxH - maxZ_P0);

                document.getElementById('lbl-base-dim').innerText = `${pLen} x ${pWid} x 170`;
                document.getElementById('lbl-limit-dim').innerText = `${pLen} x ${pWid} x ${pMaxH}`;
                
                const usedDimEl = document.getElementById('lbl-used-dim');
                usedDimEl.innerText = `${usedL.toFixed(0)} x ${usedW.toFixed(0)} x ${maxZ_P0.toFixed(0)}`;
                usedDimEl.style.color = isOverhang ? '#ef4444' : '#3b82f6';
                
                const warnEl = document.getElementById('lbl-overhang-warn');
                if (warnEl) warnEl.style.display = isOverhang ? 'inline-block' : 'none';
                
                document.getElementById('lbl-free-dim').innerText = `${freeL.toFixed(0)} x ${freeW.toFixed(0)} x ${freeH.toFixed(0)}`;
                
                // Cập nhật nhãn đếm hộp 3D
                this.updateLayerLabels();
                
                this.saveState();
            }
        };

        window.addEventListener('DOMContentLoaded', () => { app.init(); });
